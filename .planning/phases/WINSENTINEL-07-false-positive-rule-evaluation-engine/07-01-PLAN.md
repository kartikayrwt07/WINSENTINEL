from __future__ import annotations
from datetime import datetime, timezone
from typing import Any
from pydantic import BaseModel, ConfigDict, Field
from app.models.events import NormalizedEvent

class BenchmarkLabel(BaseModel):
    """Ground truth annotation metadata for benchmark events."""
    model_config = ConfigDict(frozen=True)

    is_malicious: bool = False
    expected_rules: list[str] = Field(default_factory=list)
    scenario: str = ""
    description: str = ""
    mitre_attack: list[str] = Field(default_factory=list)

class BenchmarkEvent(BaseModel):
    """Event record pairing normalized Windows event payload with ground truth label."""
    model_config = ConfigDict(frozen=True)

    event: NormalizedEvent
    label: BenchmarkLabel

class ConfusionMatrix(BaseModel):
    """Confusion matrix tallying binary classification counts."""
    tp: int = 0
    fp: int = 0
    tn: int = 0
    fn: int = 0

    @property
    def total(self) -> int:
        return self.tp + self.fp + self.tn + self.fn

    @property
    def actual_positives(self) -> int:
        return self.tp + self.fn

    @property
    def actual_negatives(self) -> int:
        return self.fp + self.tn

    @property
    def predicted_positives(self) -> int:
        return self.tp + self.fp

    @property
    def predicted_negatives(self) -> int:
        return self.tn + self.fn

class RuleEvaluationMetrics(BaseModel):
    """Calculated efficacy metrics for a single detection rule."""
    rule_id: str
    rule_title: str
    severity: str
    matrix: ConfusionMatrix
    precision: float
    recall: float
    f1_score: float
    false_positive_rate: float
    accuracy: float
    is_dead: bool = False
    is_noisy: bool = False

class TuningRecommendation(BaseModel):
    """Heuristic tuning recommendation for suboptimal detection rules."""
    rule_id: str
    rule_title: str
    issue: str  # e.g. "ZERO_COVERAGE", "HIGH_FPR", "LOW_PRECISION"
    severity: str  # "HIGH", "MEDIUM", "LOW"
    description: str
    suggested_actions: list[str] = Field(default_factory=list)
    sample_fp_events: list[dict[str, Any]] = Field(default_factory=list)

class EvaluationReport(BaseModel):
    """Global evaluation benchmark summary containing aggregated and per-rule metrics."""
    timestamp: datetime = Field(default_factory=lambda: datetime.now(timezone.utc))
    dataset_name: str
    total_events: int
    malicious_events: int
    benign_events: int
    total_rules_evaluated: int
    overall_matrix: ConfusionMatrix
    overall_precision: float
    overall_recall: float
    overall_f1_score: float
    overall_fpr: float
    overall_accuracy: float
    rule_metrics: dict[str, RuleEvaluationMetrics] = Field(default_factory=dict)
    tuning_recommendations: list[TuningRecommendation] = Field(default_factory=list)
    passed_quality_gates: bool = True

<description>
Implement mathematical confusion matrix calculation and metric synthesis in `app/evaluation/metrics.py`:
1. `calculate_metrics(matrix: ConfusionMatrix) -> tuple[float, float, float, float, float]`:
   - Returns `(precision, recall, f1_score, fpr, accuracy)`
   - Safe zero-division handling:
     - Precision: `1.0` if `(tp + fp) == 0` else `tp / (tp + fp)` (a rule that raises no alarms has 100% precision by definition)
     - Recall: `1.0` if `(tp + fn) == 0` else `tp / (tp + fn)` (if no malicious events are targeted, recall is 100%)
     - F1-Score: `0.0` if `(precision + recall) == 0.0` else `2 * (precision * recall) / (precision + recall)`
     - FPR (False Positive Rate): `0.0` if `(fp + tn) == 0` else `fp / (fp + tn)`
     - Accuracy: `0.0` if `total == 0` else `(tp + tn) / total`
2. `calculate_rule_metrics(rule_id: str, rule_title: str, severity: str, matrix: ConfusionMatrix, fpr_threshold: float = 0.05, precision_threshold: float = 0.80) -> RuleEvaluationMetrics`:
   - Calculates metrics via `calculate_metrics(matrix)`
   - Computes `is_dead = (matrix.tp == 0 and matrix.fp == 0)` (rule never fired on anything)
   - Computes `is_noisy = (fpr > fpr_threshold or (matrix.tp + matrix.fp > 0 and precision < precision_threshold))`
   - Returns fully populated `RuleEvaluationMetrics`
3. `aggregate_matrices(matrices: list[ConfusionMatrix]) -> ConfusionMatrix`:
   - Sums up tp, fp, tn, fn across multiple confusion matrices.
4. `calculate_overall_metrics(rule_metrics: dict[str, RuleEvaluationMetrics] | list[ConfusionMatrix]) -> tuple[ConfusionMatrix, float, float, float, float, float]`:
   - Aggregates overall matrix and computes global Precision, Recall, F1, FPR, and Accuracy.

Create unit tests in `tests/unit/test_confusion_matrix.py` covering:
- Standard classification scenarios with known numbers.
- Perfect detector (100% TP, 0 FP/FN -> Precision=1.0, Recall=1.0, F1=1.0, FPR=0.0, Accuracy=1.0).
- Pure noise detector (0 TP, all FP -> Precision=0.0, FPR > 0, is_noisy=True).
- Completely dead rule (0 TP, 0 FP, 0 FN, all TN -> Precision=1.0, Recall=1.0, is_dead=True).
- Zero denominator edge cases for every metric.
- Matrix aggregation across multiple rules.
</description>
