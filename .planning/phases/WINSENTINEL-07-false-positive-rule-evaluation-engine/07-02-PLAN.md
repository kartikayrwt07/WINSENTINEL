from __future__ import annotations
from pathlib import Path
from typing import Any
from app.detection.engine import DetectionEngine

from app.models.evaluation import (
    BenchmarkEvent,
    ConfusionMatrix,
    EvaluationReport,
    RuleEvaluationMetrics,
    TuningRecommendation,
)
from app.models.events import NormalizedEvent
from app.models.rule import DetectionRule

class RuleTuner:
    """Diagnoses rule health and synthesizes heuristic suppression recommendations."""

    def __init__(
        self,
        fpr_threshold: float = 0.05,
        precision_threshold: float = 0.80,
        suppression_min_freq: float = 0.70,
    ) -> None: ...

    def analyze(
        self,
        rule_metrics: dict[str, RuleEvaluationMetrics],
        fp_events_by_rule: dict[str, list[NormalizedEvent]],
    ) -> list[TuningRecommendation]: ...

    def generate_suppression_suggestions(
        self,
        rule_id: str,
        fp_events: list[NormalizedEvent],
    ) -> list[str]: ...

class EvaluationRunner:
    """Orchestrates detection rule matching over benchmark datasets and compiles reports."""

    def __init__(
        self,
        engine: DetectionEngine | None = None,
        tuner: RuleTuner | None = None,
    ) -> None: ...

    def run(
        self,
        dataset: list[BenchmarkEvent],
        dataset_name: str = "benchmark",
        rules: list[DetectionRule] | None = None,
        max_fpr: float = 0.05,
        min_precision: float = 0.80,
        fail_on_fp: bool = False,
    ) -> EvaluationReport: ...

    def run_file(
        self,
        dataset_path: Path | str,
        rules_path: Path | str | None = None,
        dataset_name: str | None = None,
        max_fpr: float = 0.05,
        min_precision: float = 0.80,
        fail_on_fp: bool = False,
    ) -> EvaluationReport: ...

<description>
Implement `EvaluationRunner` in `app/evaluation/runner.py`:
1. `EvaluationRunner.__init__(self, engine: DetectionEngine | None = None, tuner: RuleTuner | None = None)`:
   - Initializes or accepts an existing `DetectionEngine`.
   - Initializes or accepts a `RuleTuner`.
2. `run(self, dataset: list[BenchmarkEvent], dataset_name: str = "benchmark", rules: list[DetectionRule] | None = None, max_fpr: float = 0.05, min_precision: float = 0.80, fail_on_fp: bool = False) -> EvaluationReport`:
   - If `rules` provided, registers them into `engine` (clearing or creating a dedicated engine to avoid side effects).
   - Initializes per-rule confusion matrices: `{rule_id: ConfusionMatrix()}` for every loaded rule.
   - Tracks false positive event payloads per rule: `fp_events: dict[str, list[NormalizedEvent]] = defaultdict(list)`.
   - Counts malicious and benign events in the dataset.
   - Iterates through each `BenchmarkEvent`:
     - Calls `engine.evaluate_event(bm_event.event)`.
     - Extracts `fired_rule_ids = {match.rule_id for match in matches}`.
     - Extracts `expected_rule_ids = set(bm_event.label.expected_rules)`.
     - For every evaluated rule $r$:
       - If $r \in \text{fired\_rule\_ids}$ and $r \in \text{expected\_rule\_ids}$:
         $matrix[r].tp += 1$
       - If $r \in \text{fired\_rule\_ids}$ and $r \notin \text{expected\_rule\_ids}$:
         $matrix[r].fp += 1$
         $fp\_events[r].append(bm\_event.event)$
       - If $r \notin \text{fired\_rule\_ids}$ and $r \in \text{expected\_rule\_ids}$:
         $matrix[r].fn += 1$
       - If $r \notin \text{fired\_rule\_ids}$ and $r \notin \text{expected\_rule\_ids}$:
         $matrix[r].tn += 1$
   - Computes `RuleEvaluationMetrics` for each rule using `calculate_rule_metrics()`.
   - Computes aggregated `ConfusionMatrix` and global metrics using `calculate_overall_metrics()`.
   - Generates `tuning_recommendations` by passing `rule_metrics` and `fp_events` to `self.tuner.analyze()`.
   - Evaluates quality gates:
     - If `fail_on_fp` is True, `passed_quality_gates = (overall_matrix.fp == 0)`.
     - Else, `passed_quality_gates = (overall_fpr <= max_fpr and overall_precision >= min_precision)`.
   - Assembles and returns `EvaluationReport`.
3. `run_file(self, dataset_path: Path | str, rules_path: Path | str | None = None, ...)`:
   - Convenience method loading dataset using `load_benchmark_dataset(dataset_path)` and rules using `RuleLoader.load_rules_from_directory(rules_path or "rules/sysmon")`.

Create unit tests in `tests/unit/test_evaluation_runner.py`:
- Mock/synthetic detection rules with known trigger conditions.
- Test scenario where all events match expected rules (100% TP, 0 FP).
- Test scenario where benign events trigger unexpected rules (FP detection and sample collection).
- Test scenario where targeted events fail to match (FN detection).
- Test quality gate enforcement (`fail_on_fp=True` failing when FP > 0; passing when FP == 0).
- Test dataset name propagation and event counting.
</description>

<description>
Implement `RuleTuner` in `app/evaluation/tuner.py`:
1. `RuleTuner.__init__(self, fpr_threshold: float = 0.05, precision_threshold: float = 0.80, suppression_min_freq: float = 0.70)`:
   - Configures thresholds for flagging noisy rules and suppression attribute clustering.
2. `analyze(self, rule_metrics: dict[str, RuleEvaluationMetrics], fp_events_by_rule: dict[str, list[NormalizedEvent]]) -> list[TuningRecommendation]`:
   - Scans each rule metric:
     - If `metric.is_dead` ($TP = 0 \land FP = 0$):
       Emits `TuningRecommendation(issue="ZERO_COVERAGE", severity="LOW", description=f"Rule '{rule_id}' had 0 hits across benchmark corpus. Verify if target technique is obsolete or supply targeted attack telemetry.", suggested_actions=["Add unit test event to attacks.jsonl", "Review rule condition syntax"])`.
     - If `metric.false_positive_rate > self.fpr_threshold`:
       Emits `TuningRecommendation(issue="HIGH_FPR", severity="HIGH", description=f"Rule '{rule_id}' has FPR of {metric.false_positive_rate:.2%}, exceeding {self.fpr_threshold:.2%} threshold.", suggested_actions=suggestions, sample_fp_events=sample_events)`.
     - Else if `metric.precision < self.precision_threshold` and `(metric.matrix.tp + metric.matrix.fp > 0)`:
       Emits `TuningRecommendation(issue="LOW_PRECISION", severity="MEDIUM", description=f"Rule '{rule_id}' has Precision of {metric.precision:.2%}, below {self.precision_threshold:.2%} target.", suggested_actions=suggestions, sample_fp_events=sample_events)`.
3. `generate_suppression_suggestions(self, rule_id: str, fp_events: list[NormalizedEvent]) -> list[str]`:
   - Inspects FP events for common field values across attributes: `parent_image`, `user`, `image`, `current_directory`.
   - Counts occurrences of each value across the false positive events.
   - If an attribute value appears in $\ge \text{suppression\_min\_freq} \times len(fp\_events)$ (e.g. 70% of FP events have `parent_image == 'C:\\Windows\\CCM\\CcmExec.exe'`), formats a Sigma exclusion snippet:
     `filter_<field>: { <field>: '<value>' }`
   - Returns list of recommended exclusion filters.

Create unit tests in `tests/unit/test_rule_tuner.py`:
- Flagging dead rules with `ZERO_COVERAGE`.
- Flagging noisy rules with `HIGH_FPR` and `LOW_PRECISION`.
- Suppression condition extraction from multiple FP events sharing `parent_image="C:\\Windows\\CCM\\CcmExec.exe"`.
- Suppression condition extraction from multiple FP events sharing `user="NT AUTHORITY\\SYSTEM"`.
- Ensuring clean rules with 0 FP generate no noisy recommendations.
</description>
