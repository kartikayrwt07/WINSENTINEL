<objective>
Implement the multi-source persistence threat analyzer, persistence artifact models, and integrate both PowerShell and Persistence analyzers into the IngestionPipeline with SQLite alert persistence.

Purpose: Adversaries establish persistence to retain access across reboots and account changes. WinSentinel requires a specialized multi-source persistence correlation analyzer (PERS-01 through PERS-05) that monitors Registry Run/IFEO keys, Services/Drivers, Scheduled Tasks (strictly adhering to authoritative sources), WMI subscriptions, and Startup folder drops. Furthermore, both analyzers must integrate seamlessly into `IngestionPipeline` (PWSH-01..05, PERS-01..05) to persist high-confidence alerts and forensic evidence into SQLite.
Output: `app/models/persistence.py`, `app/analyzers/persistence_analyzer.py`, `app/ingestion/pipeline.py`, and comprehensive test suites in `tests/unit/test_persistence_analyzer.py` and `tests/integration/test_analyzers_pipeline_integration.py`.
</objective>

class PersistenceType(str, Enum):
    REGISTRY = "REGISTRY"
    SERVICE = "SERVICE"
    DRIVER = "DRIVER"
    SCHEDULED_TASK = "SCHEDULED_TASK"
    WMI_SUBSCRIPTION = "WMI_SUBSCRIPTION"
    STARTUP_FOLDER = "STARTUP_FOLDER"

class PersistenceArtifact(BaseModel):
    persistence_type: PersistenceType
    event_id: int
    record_id: int | None = None
    event_db_id: int | None = None
    computer: str
    timestamp_utc: datetime
    process_guid: str | None = None
    user: str | None = None
    target_name: str    # Reg key, service name, task name, consumer name, or file path
    payload: str | None = None # Binary path, command line, script block, WMI query
    severity: str       # "Low", "Medium", "High", "Critical"
    mitre_tactics: list[str]
    mitre_techniques: list[str]
    evidence_fields: dict[str, Any]
    explanation: str
    investigation_steps: list[str]
    def to_alert_dict(self) -> dict[str, Any]: ...
    def to_evidence_dicts(self, alert_id: int) -> list[dict[str, Any]]: ...

class PersistenceAnalyzer:
    def check_registry_persistence(self, event: NormalizedEvent) -> PersistenceArtifact | None: ...
    def check_service_driver_persistence(self, event: NormalizedEvent) -> PersistenceArtifact | None: ...
    def check_scheduled_task_persistence(self, event: NormalizedEvent) -> PersistenceArtifact | None: ...
    def check_wmi_persistence(self, event: NormalizedEvent) -> PersistenceArtifact | None: ...
    def check_startup_folder_persistence(self, event: NormalizedEvent) -> PersistenceArtifact | None: ...
    def analyze_event(self, event: NormalizedEvent) -> PersistenceArtifact | None: ...
</interfaces>
