class InvalidStatusTransitionError(ValueError):
    """Raised when an illegal alert status lifecycle transition is attempted."""
    pass

class AlertManager:
    VALID_TRANSITIONS: dict[str, set[str]] = {
        "New": {"Investigating"},
        "Investigating": {"Escalated", "Resolved", "False Positive"},
        "Escalated": {"Investigating", "Resolved", "False Positive"},
        "Resolved": {"Investigating"},
        "False Positive": {"Investigating"},
    }

    def __init__(self, db_manager: DatabaseManager, dedup_window_minutes: int = 60) -> None: ...
    def process_match(
        self,
        match: DetectionMatch,
        event: NormalizedEvent,
        event_db_id: int | None = None,
    ) -> tuple[int, bool]: ...  # returns (alert_id, is_duplicate)
    def process_alert(
        self,
        rule_id: str,
        title: str,
        severity: str,
        computer: str,
        user: str | None = None,
        process_guid: str | None = None,
        description: str | None = None,
        first_seen: datetime | str | None = None,
        last_seen: datetime | str | None = None,
        mitre_tactics: list[str] | None = None,
        mitre_techniques: list[str] | None = None,
        evidence_items: list[dict[str, Any]] | None = None,
    ) -> tuple[int, bool]: ...
    def update_status(
        self,
        alert_id: int,
        new_status: str,
        author: str,
        note: str | None = None,
    ) -> dict[str, Any]: ...
    def add_analyst_note(
        self,
        author: str,
        note_text: str,
        alert_id: int | None = None,
        investigation_id: int | None = None,
        disposition: str | None = None,
    ) -> int: ...
    def get_alert_forensic_drilldown(self, alert_id: int) -> dict[str, Any]: ...

class IngestionResult:
    # Existing properties + incidents and deduplicated alert metrics
    @property
    def incidents(self) -> list[CompoundIncident]: ...

    - Implement `AlertManager`:
      - `VALID_TRANSITIONS = { "New": {"Investigating"}, "Investigating": {"Escalated", "Resolved", "False Positive"}, "Escalated": {"Investigating", "Resolved", "False Positive"}, "Resolved": {"Investigating"}, "False Positive": {"Investigating"} }`.
      - `__init__(self, db_manager: DatabaseManager, dedup_window_minutes: int = 60)`:
        - Stores db_manager, alert_repo = AlertRepository(db_manager), event_repo = EventRepository(db_manager).
        - In-memory sliding cache `_window_cache: dict[tuple[str, str, str], dict[str, Any]]` tracking active alert IDs and last_seen timestamps.
      - `process_alert(...) -> tuple[int, bool]`:
        - Composite key: `(rule_id, computer, process_guid or "GLOBAL")`.
        - Checks in-memory cache or queries SQLite `alerts` table for active matching alert within `dedup_window_minutes`.
        - If duplicate:
          - Calls `alert_repo.touch_alert(alert_id, last_seen, count_increment=1)`.
          - Links any provided evidence items via `alert_repo.link_alert_evidence(...)`.
          - Updates cache.
          - Returns `(alert_id, True)`.
        - If new:
          - Calls `alert_repo.create_alert(...)`.
          - Links evidence items.
          - Updates cache.
          - Returns `(alert_id, False)`.
      - `process_match(self, match: DetectionMatch, event: NormalizedEvent, event_db_id: int | None = None) -> tuple[int, bool]`:
        - Extracts alert parameters and evidence from DetectionMatch and delegates to `process_alert`.
      - `update_status(self, alert_id: int, new_status: str, author: str, note: str | None = None) -> dict[str, Any]`:
        - Fetches current alert from `alert_repo.get_alert(alert_id)`.
        - Validates transition `current_status -> new_status` in `VALID_TRANSITIONS`. Raises `InvalidStatusTransitionError` if invalid.
        - Updates status via `alert_repo.update_alert_status(alert_id, new_status)`.
        - If `note` is provided, calls `alert_repo.add_analyst_note(author=author, note_text=note, alert_id=alert_id)`.
        - Returns updated alert dictionary.
      - `add_analyst_note(self, author: str, note_text: str, alert_id: int | None = None, investigation_id: int | None = None, disposition: str | None = None) -> int`:
        - Appends disposition to note text if provided (`f"[{disposition}] {note_text}"`).
        - Calls `alert_repo.add_analyst_note(...)`.
      - `get_alert_forensic_drilldown(self, alert_id: int) -> dict[str, Any]`:
        - Fetches alert entity with evidence and notes.
        - Fetches raw and normalized events for all evidence `event_id`s from `event_repo`.
        - If alert has `process_guid`, fetches process record and parent/child processes from `processes` table.
        - Returns structured drilldown dictionary.
